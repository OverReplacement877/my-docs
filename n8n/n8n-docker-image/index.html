    n8n docker image with native postgresql

    Step 1: Verify PostgreSQL is reachable from Docker(you can do this in pgAdmin)

    CREATE DATABASE n8n;
CREATE SCHEMA schema_n8n;
GRANT ALL ON SCHEMA schema_n8n TO postgres;

    Step 2: Create a project folder, choose any path

    mkdir n8n-docker
cd n8n-docker

    Step 3: Create docker-compose.yml and add this content

    services:
n8n:
    image: docker.n8n.io/n8nio/n8n
    container_name: n8n
    restart: unless-stopped
    ports:
    - "5678:5678"
    environment:
    - TZ=Asia/Manila
    - GENERIC_TIMEZONE=Asia/Manila

    # n8n settings
    - N8N_ENFORCE_SETTINGS_FILE_PERMISSIONS=true
    - N8N_RUNNERS_ENABLED=true

    # Database (PostgreSQL on host)
    - DB_TYPE=postgresdb
    - DB_POSTGRESDB_HOST=host.docker.internal
    - DB_POSTGRESDB_PORT=5432
    - DB_POSTGRESDB_DATABASE=n8n
    - DB_POSTGRESDB_SCHEMA=schema_n8n
    - DB_POSTGRESDB_USER=postgres
    - DB_POSTGRESDB_PASSWORD=P@ssw0rd!

    volumes:
    - n8n_data:/home/node/.n8n

volumes:
n8n_data:

    STEP 4: Start n8n

    docker compose up -d

    Check logs:

    docker compose logs -f

    You should see:

    n8n ready on 0.0.0.0, port 5678

    STEP 5: Open n8n

    Open browser:

    http://localhost:5678

    Complete the n8n setup wizard
